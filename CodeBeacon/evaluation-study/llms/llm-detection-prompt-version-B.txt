def get_db() -> psycopg.Connection:
    if "db" not in g:
        settings = current_app.config["SETTINGS"]
        g.db = psycopg.connect(
            host=settings.db_host,
            port=settings.db_port,
            dbname=settings.db_name,
            user=settings.db_user,
            password=settings.db_password,
            sslmode=settings.db_sslmode,
            connect_timeout=settings.db_connect_timeout_seconds,
            application_name="room_monitor",
            options="-c statement_timeout=15000 -c lock_timeout=5000 -c timezone=UTC",
            autocommit=True,
            row_factory=dict_row,
        )
    return g.db


def close_db(error: BaseException | None = None) -> None:
    connection = g.pop("db", None)
    if connection is not None:
        connection.close()

def create_app(settings: Settings | None = None) -> Flask:
    settings = settings if settings is not None else load_settings()
    app = Flask(__name__, static_folder=None)
    app.config.update(
        SETTINGS=settings,
        SECRET_KEY=settings.secret_key,
        MAX_CONTENT_LENGTH=16 * 1024,
        DUMMY_PASSWORD_HASH=generate_password_hash(secrets.token_urlsafe(32), method=PASSWORD_HASH_METHOD),
    )
    app.logger.setLevel(settings.log_level)
    app.teardown_appcontext(close_db)
    for blueprint in (accounts.blueprint, devices.blueprint, readings.blueprint, commands.blueprint):
        app.register_blueprint(blueprint)

    @app.before_request
    def require_https():
        command_poll = request.endpoint == "commands.pending" and request.method == "GET"
        if not request.is_secure and not command_poll:
            raise ApiError(400, "https_required", "Use HTTPS to access the API")

    @app.after_request
    def response_headers(response):
        response.headers["Cache-Control"] = "no-store"
        response.headers["X-Content-Type-Options"] = "nosniff"
        if request.is_secure:
            response.headers["Strict-Transport-Security"] = "max-age=31536000"
        if response.status_code == 401:
            response.headers["WWW-Authenticate"] = "Bearer"
        return response

    @app.errorhandler(ApiError)
    def api_error(error):
        response = app.json.response({"error": {"code": error.code, "message": error.message}})
        response.status_code = error.status
        if error.retry_after is not None:
            response.headers["Retry-After"] = str(error.retry_after)
        return response

    @app.errorhandler(HTTPException)
    def http_error(error):
        response = error.get_response()
        response.data = app.json.dumps({
            "error": {"code": error.name.lower().replace(" ", "_"), "message": error.description},
        })
        response.content_type = "application/json"
        return response

    @app.errorhandler(DatabaseError)
    def database_error(error):
        app.logger.error("Database operation failed: type=%s", type(error).__name__)
        return {"error": {"code": "database_unavailable", "message": "Storage is temporarily unavailable"}}, 503

    @app.errorhandler(Exception)
    def unexpected_error(error):
        app.logger.error("Request failed: type=%s", type(error).__name__)
        return {"error": {"code": "internal_error", "message": "The request could not be completed"}}, 500

    @app.cli.command("provision-device")
    @click.option("--username", required=True)
    @click.option("--device-id", required=True)
    @click.option("--name", required=True)
    def provision_device_command(username: str, device_id: str, name: str):
        """Assign a device with an existing credential to an account."""
        token = click.prompt("Device token", hide_input=True, confirmation_prompt=True)
        if not re.fullmatch(r"[A-Za-z0-9_-]{16,256}", token):
            raise click.ClickException("Token must contain 16 to 256 URL-safe characters")
        if not name.strip() or len(name) > 100:
            raise click.ClickException("Name must contain 1 to 100 characters")
        try:
            device = provision_device(normalize_username(username), device_id, name.strip(), token)
        except (ApiError, HTTPException) as error:
            raise click.ClickException(str(error)) from None
        except DatabaseError:
            raise click.ClickException("Storage is unavailable; check database configuration and schema") from None
        click.echo(f"Provisioned device {device['id']}")

    return app


def serve(app: Flask, settings: Settings, context: ssl.SSLContext) -> None:
    servers = []
    threads = []
    finished = Event()

    def run(server):
        try:
            server.serve_forever(poll_interval=0.2)
        finally:
            finished.set()

    try:
        servers.append(make_server(settings.host, settings.port, app, threaded=True, ssl_context=context))
        servers.append(make_server(settings.host, settings.command_poll_http_port, app, threaded=True))
        for server in servers:
            thread = Thread(target=run, args=(server,), daemon=True)
            thread.start()
            threads.append(thread)
        logging.getLogger(__name__).info(
            "API listening on HTTPS port %s and command-poll HTTP port %s",
            settings.port, settings.command_poll_http_port,
        )
        finished.wait()
        raise OSError("An API listener stopped unexpectedly")
    except KeyboardInterrupt:
        pass
    finally:
        for server in servers[:len(threads)]:
            server.shutdown()
        for thread in threads:
            thread.join()
        for server in servers:
            server.server_close()


def main() -> int:
    try:
        settings = load_settings()
        logging.basicConfig(level=settings.log_level, format="%(asctime)s %(levelname)s %(name)s %(message)s")
        if not settings.tls_cert_file.is_file() or not settings.tls_key_file.is_file():
            raise ConfigurationError("Configure existing BACKEND_TLS_CERT_FILE and BACKEND_TLS_KEY_FILE files")
        context = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
        context.minimum_version = ssl.TLSVersion.TLSv1_2
        context.load_cert_chain(settings.tls_cert_file, settings.tls_key_file)
        app = create_app(settings)
        serve(app, settings, context)
    except (ConfigurationError, OSError, ValueError) as error:
        logging.getLogger(__name__).error("Backend startup failed: %s", error)
        return 2
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

import hashlib
import hmac
import math
import re
from contextlib import contextmanager
from datetime import datetime, timedelta, timezone
from functools import wraps
from uuid import UUID

from flask import current_app, g, request
from itsdangerous import BadData, URLSafeTimedSerializer
from werkzeug.exceptions import BadRequest, Unauthorized

from backend.database import get_db

PASSWORD_HASH_METHOD = "pbkdf2:sha256:600000"
LOGIN_FAILURE_LIMIT = 5
LOGIN_WINDOW_SECONDS = 300
LOGIN_BUCKET_COUNT = 4096


class ApiError(Exception):
    def __init__(self, status: int, code: str, message: str, *, retry_after: int | None = None):
        super().__init__(message)
        self.status = status
        self.code = code
        self.message = message
        self.retry_after = retry_after


def json_body(*fields: str) -> dict:
    data = request.get_json()
    if not isinstance(data, dict) or set(data) != set(fields):
        raise BadRequest("The request must contain exactly the required JSON fields")
    return data


def bounded_text(value: object, name: str, maximum: int) -> str:
    if not isinstance(value, str) or not value or len(value) > maximum:
        raise BadRequest(f"{name} must contain 1 to {maximum} characters")
    return value


def normalize_username(value: object) -> str:
    username = bounded_text(value, "username", 32).strip().lower()
    if not re.fullmatch(r"[a-z0-9_]{3,32}", username):
        raise BadRequest("username must contain 3 to 32 letters, digits, or underscores")
    return username


def validate_password_strength(password: str) -> None:
    if not 15 <= len(password) <= 128:
        raise BadRequest("password must contain 15 to 128 characters")


def validate_device_id(value: str) -> str:
    if not re.fullmatch(r"[A-Za-z0-9_-]{1,64}", value):
        raise BadRequest("Invalid device identifier")
    return value


def uuid_value(value: object, name: str) -> UUID:
    try:
        if not isinstance(value, str):
            raise ValueError
        return UUID(value)
    except ValueError:
        raise BadRequest(f"{name} must be a UUID") from None


def fan_state(value: object) -> str:
    if value not in ("on", "off"):
        raise BadRequest("fan_state must be on or off")
    return value


def page_limit() -> int:
    try:
        limit = int(request.args.get("limit", "100"))
    except ValueError:
        raise BadRequest("limit must be an integer between 1 and 200") from None
    if not 1 <= limit <= 200:
        raise BadRequest("limit must be between 1 and 200")
    return limit


def reading_payload(data: dict) -> dict:
    reading_id = uuid_value(data["reading_id"], "reading_id")
    try:
        timestamp = datetime.fromisoformat(bounded_text(data["recorded_at"], "recorded_at", 64))
        if timestamp.tzinfo is None or timestamp.utcoffset() is None:
            raise ValueError
        timestamp = timestamp.astimezone(timezone.utc)
    except (ValueError, OverflowError):
        raise BadRequest("recorded_at must be an ISO timestamp with a timezone") from None
    if timestamp > datetime.now(timezone.utc) + timedelta(minutes=5):
        raise BadRequest("recorded_at cannot be more than five minutes in the future")
    temperature = data["temperature_c"]
    if (
        isinstance(temperature, bool) or not isinstance(temperature, (int, float))
        or not math.isfinite(temperature) or not -40 <= temperature <= 85
    ):
        raise BadRequest("temperature_c must be a finite number between -40 and 85")
    if data["unit"] != "C":
        raise BadRequest("unit must be C")
    return {
        "id": reading_id, "recorded_at": timestamp,
        "temperature_c": float(temperature), "fan_state": fan_state(data["fan_state"]),
    }


def _serializer() -> URLSafeTimedSerializer:
    return URLSafeTimedSerializer(
        current_app.config["SECRET_KEY"], salt="account-access",
        signer_kwargs={"digest_method": hashlib.sha256},
    )


def issue_account_token(account: dict) -> dict:
    return {
        "access_token": _serializer().dumps({"sub": str(account["id"]), "v": account["token_version"]}),
        "token_type": "Bearer",
        "expires_in": current_app.config["SETTINGS"].account_token_ttl_seconds,
    }


def _bearer_token() -> str:
    parts = request.headers.get("Authorization", "").split()
    if len(parts) != 2 or parts[0].lower() != "bearer" or len(parts[1]) > 2048:
        raise Unauthorized("A valid bearer token is required")
    return parts[1]


def account_required(view):
    @wraps(view)
    def wrapped(*args, **kwargs):
        try:
            payload = _serializer().loads(
                _bearer_token(), max_age=current_app.config["SETTINGS"].account_token_ttl_seconds,
            )
            account_id = UUID(payload["sub"])
            version = payload["v"]
            if type(version) is not int:
                raise ValueError
        except (BadData, ValueError, TypeError, KeyError):
            raise Unauthorized("Invalid or expired account token") from None
        account = get_db().execute(
            'SELECT id, username, token_version, created_at FROM "user" WHERE id = %s',
            (account_id,),
        ).fetchone()
        if account is None or account["token_version"] != version:
            raise Unauthorized("Invalid or expired account token")
        g.account = account
        return view(*args, **kwargs)
    return wrapped


def device_required(view):
    @wraps(view)
    def wrapped(device_id: str, *args, **kwargs):
        validate_device_id(device_id)
        token_hash = hashlib.sha256(_bearer_token().encode()).hexdigest()
        device = get_db().execute(
            "SELECT id, token_hash FROM devices WHERE id = %s", (device_id,),
        ).fetchone()
        if device is None or not hmac.compare_digest(device["token_hash"], token_hash):
            raise Unauthorized("Invalid device credential")
        return view(device_id, *args, **kwargs)
    return wrapped


@contextmanager
def login_attempt(username: str):
    digest = hmac.new(
        current_app.config["SECRET_KEY"].encode(), username.encode(), hashlib.sha256,
    ).digest()
    bucket = int.from_bytes(digest[:2]) % LOGIN_BUCKET_COUNT
    connection = get_db()
    blocked_for = None
    result = {"authenticated": False}
    with connection.transaction():
        connection.execute(
            "INSERT INTO login_attempts (bucket, failures, window_started_at) "
            "VALUES (%s, 0, clock_timestamp()) ON CONFLICT (bucket) DO NOTHING", (bucket,),
        )
        attempt = connection.execute(
            "SELECT failures, window_started_at FROM login_attempts WHERE bucket = %s FOR UPDATE",
            (bucket,),
        ).fetchone()
        now = connection.execute("SELECT clock_timestamp() AS now").fetchone()["now"]
        remaining = LOGIN_WINDOW_SECONDS - (now - attempt["window_started_at"]).total_seconds()
        failures = attempt["failures"]
        if remaining <= 0:
            failures = 0
            connection.execute(
                "UPDATE login_attempts SET failures = 0, window_started_at = %s WHERE bucket = %s",
                (now, bucket),
            )
        elif failures >= LOGIN_FAILURE_LIMIT:
            blocked_for = max(1, math.ceil(remaining))
        if blocked_for is None:
            yield result
            connection.execute(
                "UPDATE login_attempts SET failures = %s WHERE bucket = %s",
                (0 if result["authenticated"] else min(failures + 1, LOGIN_FAILURE_LIMIT), bucket),
            )
    if blocked_for is not None:
        raise ApiError(429, "login_limited", "Too many login attempts; try again later", retry_after=blocked_for)


def public_row(row: dict) -> dict:
    return {
        key: value.astimezone(timezone.utc).isoformat() if isinstance(value, datetime)
        else str(value) if isinstance(value, UUID) else value
        for key, value in row.items()
    }

from flask import current_app
from psycopg.errors import UniqueViolation
from werkzeug.exceptions import Unauthorized

from backend.database import get_db
from backend.security import ApiError, PASSWORD_HASH_METHOD, validate_password_strength


def register_account(username: str, password: str) -> dict:
    validate_password_strength(password)
    password_hash = generate_password_hash(password, method=PASSWORD_HASH_METHOD)
    try:
        return get_db().execute(
            'INSERT INTO "user" (id, username, password_hash) VALUES (%s, %s, %s) '
            "RETURNING id, username, token_version, created_at",
            (uuid4(), username, password_hash),
        ).fetchone()
    except UniqueViolation:
        raise ApiError(409, "username_taken", "That username is already registered") from None


def authenticate(username: str, password: str) -> dict | None:
    account = get_db().execute(
        'SELECT id, username, password_hash, token_version, created_at FROM "user" WHERE username = %s',
        (username,),
    ).fetchone()
    password_hash = account["password_hash"] if account else current_app.config["DUMMY_PASSWORD_HASH"]
    valid = check_password_hash(password_hash, password)
    if account is None or not valid:
        return None
    return {key: value for key, value in account.items() if key != "password_hash"}


def change_password(account_id, current_password: str, new_password: str) -> dict:
    validate_password_strength(new_password)
    connection = get_db()
    with connection.transaction():
        account = connection.execute(
            'SELECT password_hash FROM "user" WHERE id = %s FOR UPDATE', (account_id,),
        ).fetchone()
        if account is None or not check_password_hash(account["password_hash"], current_password):
            raise Unauthorized("Current password is incorrect")
        password_hash = generate_password_hash(new_password, method=PASSWORD_HASH_METHOD)
        return connection.execute(
            'UPDATE "user" SET password_hash = %s, token_version = token_version + 1 WHERE id = %s '
            "RETURNING id, username, token_version, created_at", (password_hash, account_id),
        ).fetchone()

            return row, True

            "VALUES (%s, %s, %s, %s, clock_timestamp()) "
            f"RETURNING {COMMAND_COLUMNS}", (uuid4(), device_id, requested_by, state),
        ).fetchone()

def acknowledge_command(device_id: str, command_id, state: str) -> dict:
    connection = get_db()
    with connection.transaction():
        connection.execute("SELECT id FROM devices WHERE id = %s FOR UPDATE", (device_id,))
        command = connection.execute(
            f"SELECT {COMMAND_COLUMNS} FROM requests WHERE device_id = %s AND id = %s FOR UPDATE",
            (device_id, command_id),
        ).fetchone()
        if command is None:
            raise NotFound("Command not found")
        if command["fan_state"] != state:
            raise ApiError(409, "state_mismatch", "Reported fan state does not match the command")
        if command["acknowledged_at"] is not None:
            return command
        oldest = pending_command(device_id)
        if oldest is None or oldest["id"] != command_id:
            raise ApiError(409, "command_order", "An earlier command is still pending")
        return connection.execute(
            "UPDATE requests SET acknowledged_at = clock_timestamp() WHERE device_id = %s AND id = %s "
            f"RETURNING {COMMAND_COLUMNS}", (device_id, command_id),
        ).fetchone()

class ApiError(RuntimeError):
    def __init__(
        self, reason: str, *, status_code: int | None = None,
        retryable: bool = False, certificate_error: bool = False,
    ) -> None:
        super().__init__(reason)
        self.reason = reason
        self.status_code = status_code
        self.retryable = retryable
        self.certificate_error = certificate_error


@dataclass(frozen=True, slots=True)
class Reading:
    reading_id: str
    recorded_at: str
    temperature_c: float
    fan_state: FanState

    def payload(self) -> dict[str, Any]:
        return {
            "reading_id": self.reading_id,
            "recorded_at": self.recorded_at,
            "temperature_c": self.temperature_c,
            "unit": "C",
            "fan_state": self.fan_state.value,
        }

class FanCommand:

def _is_certificate_error(error: BaseException) -> bool:
    pending = [error]
    visited: set[int] = set()
    while pending:
        current = pending.pop()
        if id(current) in visited:
            continue
        visited.add(id(current))
        if isinstance(current, ssl.SSLCertVerificationError):
            return True
        candidates = (*current.args, current.__cause__, current.__context__, getattr(current, "reason", None))
        pending.extend(item for item in candidates if isinstance(item, BaseException))
    return False


class DeviceApiClient:
    def __init__(self, settings: Settings) -> None:
        self._settings = settings
        self._session = requests.Session()
        self._session.trust_env = False
        self._headers = {
            "Authorization": f"Bearer {settings.device_token}",
            "Accept": "application/json",
        }
        self._device_path = f"/api/devices/{settings.device_id}"

    def __enter__(self) -> "DeviceApiClient":
        return self

    def __exit__(self, *args: Any) -> None:
        self._session.close()

    def _exchange(
        self, method: str, path: str, *, expected: tuple[int, ...],
        payload: dict[str, Any] | None = None, parse_json: bool = False,
        base_url: str | None = None,
    ) -> Any:
        try:
            with self._session.request(
                method, (base_url or self._settings.api_base_url) + path,
                headers=self._headers, json=payload, verify=str(self._settings.api_ca_bundle),
                timeout=(self._settings.connect_timeout_seconds, self._settings.read_timeout_seconds),
                allow_redirects=False,
            ) as response:
                if response.status_code not in expected:
                    raise ApiError(
                        "unexpected_response", status_code=response.status_code,
                        retryable=response.status_code in {408, 429} or response.status_code >= 500,
                    )
                if parse_json:
                    try:
                        return response.json()
                    except ValueError:
                        raise ApiError("invalid_json") from None
        except requests.RequestException as error:
            certificate_error = _is_certificate_error(error)
            tls_error = isinstance(error, requests.exceptions.SSLError)
            raise ApiError(
                "certificate_validation" if certificate_error else "tls_failure" if tls_error else "transport_failure",
                retryable=not tls_error, certificate_error=certificate_error,
            ) from None

    def _post_reading(self, reading: Reading) -> None:
        try:
            self._exchange(
                "POST", self._device_path + "/readings",
                expected=(200, 201), payload=reading.payload(),
            )
        except ApiError as error:
            logger.error(
                "Reading submission failed: reason=%s status=%s",
                error.reason, error.status_code,
            )
            raise

    def submit_reading(self, reading: Reading) -> None:
        self._post_reading(reading)

    def poll_command(self) -> FanCommand | None:
        data = self._exchange(
            "GET", self._device_path + "/commands/pending",
            expected=(200,), parse_json=True, base_url=self._settings.command_poll_base_url,
        )
        if not isinstance(data, dict) or "command" not in data:
            raise ApiError("invalid_command_response")
        command = data["command"]
        if command is None:
            return None
        try:
            if not isinstance(command, dict) or not isinstance(command.get("id"), str):
                raise ValueError
            return FanCommand(str(UUID(command["id"])), FanState(command["fan_state"]))
        except (KeyError, ValueError, TypeError, AttributeError):
            raise ApiError("invalid_command_response") from None

    def acknowledge_command(self, command: FanCommand, reported_state: FanState) -> None:
        self._exchange(
            "POST", self._device_path + f"/commands/{command.command_id}/ack",
            expected=(200, 204),
            payload={"fan_state": reported_state.value},
        )

from firmware.api_client import ApiError, DeviceApiClient, FanCommand, Reading

from firmware.temperature_sensor import TemperatureSensor

logger = logging.getLogger(__name__)


class DeviceRuntime:
    def __init__(
        self, settings: Settings, client: DeviceApiClient, *,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        self._settings = settings
        self._client = client
        self._clock = clock
        self._fan = Fan()
        self._sensor = TemperatureSensor(
            seed=settings.sensor_seed,
            start_temperature_c=settings.start_temperature_c,
            ambient_temperature_c=settings.ambient_temperature_c,
            fan_cooling_c_per_second=settings.fan_cooling_c_per_second,
            noise_stddev_c=settings.noise_stddev_c,
            thermal_time_constant_seconds=settings.thermal_time_constant_seconds,
        )
        self._last_update = self._next_reading = self._next_command = clock()
        self._pending_reading: Reading | None = None
        self._pending_command: FanCommand | None = None

    def _advance_sensor(self) -> None:
        now = self._clock()
        self._sensor.advance(now - self._last_update, fan_on=self._fan.is_on)
        self._last_update = now

    def _send_reading(self) -> None:
        if self._pending_reading is None:
            self._pending_reading = Reading(
                reading_id=str(uuid4()),
                recorded_at=datetime.now(UTC).isoformat(timespec="milliseconds"),
                temperature_c=self._sensor.read_celsius(),
                fan_state=self._fan.state,
            )
        delay = self._settings.reading_interval_seconds
        try:
            self._client.submit_reading(self._pending_reading)
        except ApiError as error:
            if not error.retryable:
                raise
            delay = self._settings.reconnect_delay_seconds
            logger.warning("Reading delivery deferred: reason=%s status=%s", error.reason, error.status_code)
        else:
            logger.info("Reading delivered: temperature_c=%.2f fan=%s", self._pending_reading.temperature_c, self._pending_reading.fan_state)
            self._pending_reading = None
        self._next_reading = self._clock() + delay

    def _process_command(self) -> None:
        delay = self._settings.command_poll_interval_seconds
        try:
            if self._pending_command is None:
                command = self._client.poll_command()
                if command is not None:
                    self._advance_sensor()
                    self._fan.set_state(command.fan_state)
                    self._pending_command = command
                    logger.info("Fan command applied: command=%s state=%s", command.command_id, self._fan.state)
            if self._pending_command is not None:
                self._client.acknowledge_command(self._pending_command, self._fan.state)
                logger.info("Fan command acknowledged: command=%s", self._pending_command.command_id)
                self._pending_command = None
        except ApiError as error:
            if not error.retryable:
                raise
            delay = self._settings.reconnect_delay_seconds
            logger.warning("Command exchange deferred: reason=%s status=%s", error.reason, error.status_code)
        self._next_command = self._clock() + delay

    def step(self) -> float:
        self._advance_sensor()
        if self._clock() >= self._next_reading:
            self._send_reading()
        if self._clock() >= self._next_command:
            self._process_command()
        return min(self._next_reading, self._next_command)

    def stop(self) -> None:
        self._fan.set_state(FanState.OFF)


def configure_logging(settings: Settings) -> None:
    settings.log_file.parent.mkdir(parents=True, exist_ok=True)
    file_handler = RotatingFileHandler(
        settings.log_file, maxBytes=1_000_000, backupCount=3, encoding="utf-8",
    )
    logging.basicConfig(
        level=settings.log_level,
        format="%(asctime)s %(levelname)s %(name)s %(message)s",
        handlers=[logging.StreamHandler(), file_handler], force=True,
    )


def main() -> int:
    parser = argparse.ArgumentParser(description="Run the room monitor device client.")
    parser.add_argument("--check-config", action="store_true", help="Validate local settings without contacting the API.")
    args = parser.parse_args()
    try:
        settings = load_settings()
    except ConfigurationError as error:
        print(f"Invalid firmware configuration: {error}", file=sys.stderr)
        return 2
    except OSError:
        print("Cannot read firmware configuration. Check .env and certificate file permissions.", file=sys.stderr)
        return 2
    if args.check_config:
        print("Firmware configuration is valid; API connectivity and enrollment have not been checked.")
        return 0
    try:
        configure_logging(settings)
    except OSError:
        print("Cannot open FIRMWARE_LOG_FILE. Check the path and write permissions.", file=sys.stderr)
        return 2

    stop_event = threading.Event()

    def request_stop(signum: int, frame: object) -> None:
        stop_event.set()

    previous_handlers = {signum: signal.signal(signum, request_stop) for signum in (signal.SIGINT, signal.SIGTERM)}
    try:
        with DeviceApiClient(settings) as client:
            runtime = DeviceRuntime(settings, client)
            logger.info("Device started: device=%s", settings.device_id)
            try:
                while not stop_event.is_set():
                    deadline = runtime.step()
                    stop_event.wait(max(0, deadline - time.monotonic()))
            finally:
                runtime.stop()
                logger.info("Device stopped; fan state is off")
    except ApiError as error:
        logger.error("Device stopped after API rejection: reason=%s status=%s", error.reason, error.status_code)
        return 1
    finally:
        for signum, previous in previous_handlers.items():
            signal.signal(signum, previous)
        logging.shutdown()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

            valid_address = (
                address.scheme == scheme and address.hostname
                and address.username is None and address.password is None
                and not address.query and not address.fragment and not address.path
                and (port is None or port > 0)
            )
        except ValueError:
            valid_address = False
        if not valid_address or any(character.isspace() for character in value):
            raise ConfigurationError(f"{name} must be an {scheme.upper()} origin without credentials or a path")
        return value

    base_url = origin("API_BASE_URL", "https://localhost:8444", "https")
    command_poll_base_url = origin("DEVICE_COMMAND_POLL_BASE_URL", "http://127.0.0.1:8081", "http")

    if not ca_bundle.is_file():

import math
import random


class TemperatureSensor:
    def __init__(
        self,
        *,
        seed: int,
        start_temperature_c: float,
        ambient_temperature_c: float,
        fan_cooling_c_per_second: float,
        noise_stddev_c: float,
        thermal_time_constant_seconds: float,
    ) -> None:
        values = (
            start_temperature_c, ambient_temperature_c,
            fan_cooling_c_per_second, noise_stddev_c,
            thermal_time_constant_seconds,
        )
        if not all(math.isfinite(value) for value in values):
            raise ValueError("Sensor parameters must be finite")
        if fan_cooling_c_per_second < 0 or noise_stddev_c < 0:
            raise ValueError("Cooling and measurement noise cannot be negative")
        if thermal_time_constant_seconds <= 0:
            raise ValueError("Thermal time constant must be positive")
        self._temperature_c = start_temperature_c
        self._ambient_c = ambient_temperature_c
        self._cooling = fan_cooling_c_per_second
        self._noise = noise_stddev_c
        self._time_constant = thermal_time_constant_seconds
        self._random = random.Random(seed)

    def advance(self, elapsed_seconds: float, *, fan_on: bool) -> None:
        if not math.isfinite(elapsed_seconds) or elapsed_seconds < 0:
            raise ValueError("Elapsed time must be finite and nonnegative")
        equilibrium = self._ambient_c
        if fan_on:
            equilibrium -= self._cooling * self._time_constant
        fraction = -math.expm1(-elapsed_seconds / self._time_constant)
        self._temperature_c += (equilibrium - self._temperature_c) * fraction

    def read_celsius(self) -> float:
        return round(self._temperature_c + self._random.gauss(0, self._noise), 2)

class ConfigurationError(ValueError):
    pass


class ApiError(RuntimeError):
    def __init__(
        self, message: str, *, status_code: int | None = None,
        retry_after: int | None = None,
    ) -> None:
        super().__init__(message)
        self.status_code = status_code
        self.retry_after = retry_after


@dataclass(frozen=True, slots=True)
class Settings:
    api_base_url: str
    api_ca_bundle: Path
    connect_timeout_seconds: float
    read_timeout_seconds: float


def load_settings() -> Settings:
    values = {**dotenv_values(PROJECT_ROOT / ".env", interpolate=False), **os.environ}

    def text(name: str, default: str) -> str:
        return (values.get(name) or default).strip()

    def timeout(name: str, default: str) -> float:
        try:
            value = float(text(name, default))
        except ValueError:
            raise ConfigurationError(f"{name} must be a number") from None
        if not math.isfinite(value) or not 0.1 <= value <= 300:
            raise ConfigurationError(f"{name} must be between 0.1 and 300")
        return value

    base_url = text("API_BASE_URL", "https://localhost:8444").rstrip("/")
    try:
        address = urlsplit(base_url)
        port = address.port
        valid = (
            address.scheme == "https" and address.hostname
            and address.username is None and address.password is None
            and not address.path and not address.query and not address.fragment
            and (port is None or port > 0)
        )
    except ValueError:
        valid = False
    if not valid or any(character.isspace() for character in base_url):
        raise ConfigurationError("API_BASE_URL must be an HTTPS origin without credentials or a path")

    ca_bundle = (PROJECT_ROOT / Path(text("API_CA_BUNDLE", ".local/tls/server.crt")).expanduser()).resolve()
    if not ca_bundle.is_file():
        raise ConfigurationError("API_CA_BUNDLE must name an existing certificate file")
    return Settings(
        api_base_url=base_url,
        api_ca_bundle=ca_bundle,
        connect_timeout_seconds=timeout("API_CONNECT_TIMEOUT_SECONDS", "3"),
        read_timeout_seconds=timeout("API_READ_TIMEOUT_SECONDS", "10"),
    )


class ApiClient:
    def __init__(self, settings: Settings, access_token: str | None = None) -> None:
        self._settings = settings
        self._access_token = access_token
        self._session = requests.Session()
        self._session.trust_env = False

    def __enter__(self) -> "ApiClient":
        return self

    def __exit__(self, *args: Any) -> None:
        self._session.close()

    def _request(
        self, method: str, path: str, *, payload: dict | None = None,
        params: dict | None = None, authenticated: bool = True,
    ) -> dict:
        headers = {"Accept": "application/json"}
        if authenticated:
            if not self._access_token:
                raise ApiError("Sign in to continue.", status_code=401)
            headers["Authorization"] = f"Bearer {self._access_token}"
        try:
            with self._session.request(
                method, self._settings.api_base_url + path,
                json=payload, params=params, headers=headers,
                verify=str(self._settings.api_ca_bundle),
                timeout=(self._settings.connect_timeout_seconds, self._settings.read_timeout_seconds),
                allow_redirects=False,
            ) as response:
                if 300 <= response.status_code < 400:
                    raise ApiError("The API returned an unexpected redirect.", status_code=response.status_code)
                try:
                    data = response.json()
                except ValueError:
                    raise ApiError("The API returned an invalid response.", status_code=response.status_code) from None
                if not isinstance(data, dict):
                    raise ApiError("The API returned an invalid response.", status_code=response.status_code)
                if not 200 <= response.status_code < 300:
                    error = data.get("error")
                    message = error.get("message") if isinstance(error, dict) else None
                    if not isinstance(message, str) or not message or len(message) > 300:
                        message = "The request could not be completed. Please try again later."
                    retry = response.headers.get("Retry-After", "")
                    retry_after = int(retry) if retry.isascii() and retry.isdigit() and len(retry) <= 6 else None
                    raise ApiError(message, status_code=response.status_code, retry_after=retry_after)
                return data
        except requests.exceptions.SSLError:
            raise ApiError("The API certificate could not be verified. Check API_CA_BUNDLE and the server certificate.") from None
        except requests.exceptions.Timeout:
            raise ApiError("The API request timed out. Refresh to check its outcome before submitting again.") from None
        except requests.exceptions.RequestException:
            raise ApiError("The API could not be reached. Check that the service is running, then refresh.") from None

    @staticmethod
    def _device_path(device_id: str) -> str:
        if not re.fullmatch(r"[A-Za-z0-9_-]{1,64}", device_id):
            raise ApiError("Invalid device identifier.")
        return f"/api/devices/{device_id}"

    def register(self, username: str, password: str) -> dict:
        return self._request("POST", "/api/accounts", payload={"username": username, "password": password}, authenticated=False)

    def login(self, username: str, password: str) -> dict:
        return self._request("POST", "/api/accounts/login", payload={"username": username, "password": password}, authenticated=False)

    def current_account(self) -> dict:
        return self._request("GET", "/api/accounts/me")["account"]

    def change_password(self, current_password: str, new_password: str) -> dict:
        return self._request("PATCH", "/api/accounts/password", payload={
            "current_password": current_password, "new_password": new_password,
        })

    def enroll_device(self, name: str) -> dict:
        return self._request("POST", "/api/devices", payload={"name": name})

    def list_devices(self) -> list[dict]:
        return self._request("GET", "/api/devices", params={"limit": 200})["devices"]

    def device_detail(self, device_id: str) -> dict:
        return self._request("GET", self._device_path(device_id))

    def readings(self, device_id: str, limit: int = 100) -> list[dict]:
        return self._request("GET", self._device_path(device_id) + "/readings", params={"limit": limit})["readings"]

    def commands(self, device_id: str, limit: int = 100) -> list[dict]:
        return self._request("GET", self._device_path(device_id) + "/commands", params={"limit": limit})["commands"]

    def request_fan(self, device_id: str, fan_state: str) -> dict:
        if fan_state not in {"on", "off"}:
            raise ApiError("Fan state must be on or off.")
        return self._request("POST", self._device_path(device_id) + "/commands", payload={"fan_state": fan_state})["command"]

import time

import streamlit as st

from frontend.api_client import ApiClient, ApiError


def clear_session(message: str | None = None) -> None:
    for key in list(st.session_state):
        del st.session_state[key]
    if message:
        st.session_state["notice"] = message


def save_session(response: dict) -> None:
    st.session_state["auth"] = {
        "access_token": response["access_token"],
        "expires_at": time.monotonic() + response["expires_in"],
        "account": response["account"],
    }


def require_session() -> dict:
    auth = st.session_state.get("auth")
    if not auth or time.monotonic() >= auth["expires_at"]:
        clear_session("Your session has expired. Sign in again.")
        st.rerun(scope="app")
    return auth


def show_api_error(error: ApiError) -> None:
    st.error(str(error))
    if error.retry_after is not None:
        st.caption(f"Try again in {error.retry_after} seconds.")


def handle_authenticated_error(error: ApiError) -> None:
    if error.status_code == 401:
        clear_session("Your session is no longer valid. Sign in again.")
        st.rerun(scope="app")
    show_api_error(error)


def render_sign_in(client: ApiClient) -> None:
    st.caption("Monitor room temperatures and manage your devices.")
    mode = st.segmented_control("Account access", ["Sign in", "Create account"], default="Sign in", key="access_mode")
    registering = mode == "Create account"
    with st.form("account_access", clear_on_submit=True):
        st.subheader("Create account" if registering else "Sign in")
        username = st.text_input("Username", max_chars=32, key="login_username")
        password = st.text_input("Password", type="password", max_chars=128, key="login_password")
        confirmation = st.text_input("Confirm password", type="password", max_chars=128, key="register_confirmation") if registering else None
        if registering:
            st.caption("Use 3–32 letters, numbers, or underscores for your username and 15–128 characters for your password.")
        submitted = st.form_submit_button("Create account" if registering else "Sign in", type="primary", width="stretch")
    if not submitted:
        return
    if not username.strip() or not password:
        st.error("Enter your username and password.")
        return
    if registering and password != confirmation:
        st.error("Passwords do not match.")
        return
    try:
        if registering:
            client.register(username, password)
            st.success("Account created. Sign in to continue.")
        else:
            response = client.login(username, password)
            clear_session()
            save_session(response)
            st.rerun()
    except ApiError as error:
        show_api_error(error)

    st.header("Account settings")
    st.text(f"Username: {st.session_state['auth']['account']['username']}")
    with st.form("change_password", clear_on_submit=True):
        st.subheader("Change password")
        st.caption("Use 15 to 128 characters for your new password.")
        current = st.text_input("Current password", type="password", max_chars=128, key="current_password")
        new = st.text_input("New password", type="password", max_chars=128, key="new_password")
        confirmation = st.text_input("Confirm new password", type="password", max_chars=128, key="confirm_new_password")
        submitted = st.form_submit_button("Update password", type="primary")
    if submitted:
        if not current or not new:
            st.error("Enter your current and new passwords.")
        elif new != confirmation:
            st.error("Passwords do not match.")
        else:
            try:
                save_session(client.change_password(current, new))
                st.success("Password updated. Other account sessions have been signed out.")
            except ApiError as error:
                show_api_error(error)
